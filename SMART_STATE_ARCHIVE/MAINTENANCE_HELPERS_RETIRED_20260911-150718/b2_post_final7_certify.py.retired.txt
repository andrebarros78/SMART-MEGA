from __future__ import annotations
import hashlib, json, os, pathlib, shutil

root=pathlib.Path.cwd()

archive.mkdir(parents=True,exist_ok=True)

def sha(p):
 h=hashlib.sha256()
 with pathlib.Path(p).open('rb') as f:
  for c in iter(lambda:f.read(1024*1024),b''): h.update(c)
 return h.hexdigest()

def backup(p):
 p=pathlib.Path(p)
 if not p.exists(): return None
 d=sha(p); dest=archive/(p.name+'.'+d+'.bak')
 if not dest.exists(): shutil.copy2(p,dest)
 if sha(dest)!=d: raise RuntimeError('backup hash mismatch: '+p.name)
 return str(dest.relative_to(root)).replace('\\','/')

def atomic_bytes(p,data):
 p=pathlib.Path(p); tmp=p.with_name(p.name+'.final7.tmp'); tmp.write_bytes(data); os.replace(tmp,p)

def atomic_json(p,obj):
 atomic_bytes(p,(json.dumps(obj,ensure_ascii=False,indent=2)+'\n').encode('utf-8'))
