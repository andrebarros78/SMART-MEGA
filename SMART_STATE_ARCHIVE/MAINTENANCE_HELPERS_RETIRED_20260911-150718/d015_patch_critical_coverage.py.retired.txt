from pathlib import Path
import hashlib, json

root = Path(__file__).resolve().parent
changes = {}

# 1) Make duplicate detection reachable before contiguity detection.
p = root / 'smart_canonical_history_advance.py'
raw = p.read_bytes(); before = hashlib.sha256(raw).hexdigest(); text = p.read_text(encoding='utf-8')
old = '''    if not values or values != list(range(values[0], values[-1] + 1)):\n        raise ValueError("history contest sequence is not contiguous")\n    if len(values) != len(set(values)):\n        raise ValueError("history contains duplicate contests")\n'''
new = '''    if len(values) != len(set(values)):\n        raise ValueError("history contains duplicate contests")\n    if not values or values != list(range(values[0], values[-1] + 1)):\n        raise ValueError("history contest sequence is not contiguous")\n'''
if old not in text:
    raise SystemExit('D015_SEQUENCE_CONTEXT_NOT_FOUND')
bdir = root / 'SMART_STATE_ARCHIVE' / 'b2-evolution-20260909'; bdir.mkdir(parents=True, exist_ok=True)
bak = bdir / f'{p.name}.{before}.bak'
if not bak.exists(): bak.write_bytes(raw)
text = text.replace(old, new, 1); p.write_text(text, encoding='utf-8'); after = hashlib.sha256(p.read_bytes()).hexdigest()
changes[p.name] = {'before': before, 'after': after, 'backup': str(bak.relative_to(root)).replace('\\','/')}

p = root / 'smart_global_integrity.py'
