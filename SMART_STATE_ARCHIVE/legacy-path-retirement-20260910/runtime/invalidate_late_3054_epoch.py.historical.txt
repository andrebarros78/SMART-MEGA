from __future__ import annotations
import hashlib, json, os, shutil, tempfile
from datetime import datetime
from pathlib import Path
from smart_target_schedule import resolve_target_schedule
from smart_procedural_actionability_gate import evaluate_freeze_window

ROOT=Path(r'C:\New Projet\SMART-MEGA').resolve()
ACTIVE=ROOT/'SMART_ACTIVE_EPOCH.json'

def sha(path:Path)->str:
    h=hashlib.sha256()
    with path.open('rb') as f:
        for c in iter(lambda:f.read(1024*1024),b''): h.update(c)
    return h.hexdigest()

def sha_json(obj)->str:
    return hashlib.sha256(json.dumps(obj,ensure_ascii=False,sort_keys=True,separators=(',',':')).encode()).hexdigest()

def write_json(path:Path,obj:dict):
    path.parent.mkdir(parents=True,exist_ok=True)
    fd,tmp=tempfile.mkstemp(prefix=path.name+'.',suffix='.tmp',dir=path.parent); os.close(fd)
    t=Path(tmp)
    try:
        t.write_text(json.dumps(obj,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
        os.replace(t,path)
    finally:
        if t.exists(): t.unlink()

def main():
    active=json.loads(ACTIVE.read_text(encoding='utf-8-sig'))
    if active.get('status')!='FROZEN' or int(active.get('target_contest',-1))!=3054:
        raise SystemExit('expected current FROZEN 3054 epoch')
    freeze_ref=active.get('freeze_authority') or {}
    fp=(ROOT/str(freeze_ref.get('file',''))).resolve()
    if not fp.is_file() or sha(fp)!=freeze_ref.get('sha256'):
        raise SystemExit('active freeze hash mismatch')
    freeze=json.loads(fp.read_text(encoding='utf-8-sig'))
    schedule=resolve_target_schedule(ROOT,3054)
    observed=str(freeze.get('frozen_utc',''))
    verdict=evaluate_freeze_window(scheduled_datetime=str(schedule['scheduled_datetime']),observed_at=observed)
    if verdict.freeze_authority_allowed:
        raise SystemExit('freeze is not late; refusing invalidation')
    proof={
      'schema':'SMART_TEMPORAL_FREEZE_INVALIDATION_1',
      'status':'PASS_INVALIDATED_PROSPECTIVE_AUTHORITY',
      'project':'SMART-MEGA',
      'epoch_id':active['epoch_id'],
      'active_epoch_sha256_before':active['sha256'],
      'target_contest':3054,
      'cutoff_contest':3053,
      'freeze_file':str(fp.relative_to(ROOT)).replace('\\','/'),
      'freeze_sha256':sha(fp),
      'frozen_utc':observed,
      'scheduled_datetime':schedule['scheduled_datetime'],
      'schedule_authority':schedule,
      'freeze_window_verdict':verdict.to_dict(),
      'reason':'FREEZE_CREATED_AFTER_AUTHORITATIVE_SCHEDULED_DRAW_START',
      'prospective_scientific_eligibility':False,
      'engineering_generation_authority_for_3054':False,
      'result_numbers_used_to_determine_invalidation':False,
      'artifact_preserved':True,
      'retroactive_rewrite':False,
    }
    proof['sha256']=sha_json(proof)
    epoch_dir=fp.parent
    epoch_proof=epoch_dir/'TEMPORAL_INVALIDATION_PROOF.json'
    root_proof=ROOT/'SMART_3054_TEMPORAL_INVALIDATION_PROOF.json'
    write_json(epoch_proof,proof); write_json(root_proof,proof)
    if sha(epoch_proof)!=sha(root_proof): raise SystemExit('invalidation proof copies differ')
    archive=ROOT/'SMART_STATE_ARCHIVE'/f"active_epoch_temporally_invalid_{sha(ACTIVE)}.json"
    archive.parent.mkdir(parents=True,exist_ok=True)
    if not archive.exists(): shutil.copy2(ACTIVE,archive)
    if sha(archive)!=sha(ACTIVE): raise SystemExit('active epoch archive hash mismatch')
    ACTIVE.unlink()
    print(json.dumps({'status':'PASS','invalidation_proof':str(root_proof),'invalidation_proof_sha256':sha(root_proof),'freeze_sha256':sha(fp),'scheduled_datetime':schedule['scheduled_datetime'],'frozen_utc':observed,'active_pointer_removed':not ACTIVE.exists(),'archive':str(archive)},ensure_ascii=False,indent=2))
if __name__=='__main__': main()
